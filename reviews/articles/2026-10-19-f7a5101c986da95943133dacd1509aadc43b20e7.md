### Reproduce the coordinator’s budget boundary controls

In a separate checkout at the exact source revision f7a5101, install with `pnpm install --frozen-lockfile`. Save this code as `review-budget-control.mjs` at its root and run `pnpm exec node --import tsx review-budget-control.mjs`. The injected agents count calls without contacting a provider. Registry changes remain in memory. This control was run by the separate coordinator validation lane, not by the original scheduled reviewer.

```javascript
import assert from 'node:assert/strict'
import { execFileSync } from 'node:child_process'
import { readActionRegistry, readArticleStatuses } from './src/action-registry.ts'
import { reviewArticleActions, reviewDraftActions } from './src/review-actions.ts'

const repo = process.cwd()
const pin = 'f7a5101c986da95943133dacd1509aadc43b20e7'
assert.equal(execFileSync('git', ['rev-parse', 'HEAD'], { cwd: repo, encoding: 'utf8' }).trim(), pin)
const markdown = execFileSync('git', ['show', `${pin}:_articles/2026-10-19.md`], { cwd: repo, encoding: 'utf8' })
const registry = readActionRegistry(repo), articleStatuses = readArticleStatuses(repo)
const before = JSON.stringify(registry), results = []
const source = (id, content) => ({ id, kind: 'repository', locator: 'Synthetic offline budget counterexample', content })
for (const mode of ['single-source', 'aggregate-sources', 'proposer-prompt', 'verifier-prompt']) {
  let proposer = 0, verifier = 0
  const value = structuredClone(registry)
  const sources = mode === 'single-source' ? [source('bytes-not-characters', '€'.repeat(43691))]
    : mode === 'aggregate-sources' ? Array.from({ length: 4 }, (_, i) => source(`source-${i}`, 'x'.repeat(100 * 1024)))
      : [source('small-source', 'Synthetic exact source evidence for an offline control.')]
  if (mode === 'proposer-prompt') value.records[0].workKey = 'x'.repeat(800000)
  const agents = {
    identity: { proposer: 'offline-counter', verifier: 'offline-counter' },
    propose: async () => {
      proposer++
      return mode === 'verifier-prompt' ? {
        changes: [{ actionKey: value.records[0].key, state: 'open', description: 'x'.repeat(600000), reason: 'Synthetic oversized verifier input only.', citations: [{ sourceId: 's1', quote: sources[0].content }] }],
        suggestions: [],
      } : { changes: [], suggestions: [] }
    },
    verify: async () => { verifier++; throw new Error('Verifier must not be called by these controls') },
  }
  const input = { registry: value, articleDate: '2026-10-19', markdown, feedback: 'Offline review boundary control; no publication or execution.', sources, agents, articleStatuses }
  let error
  try { await reviewArticleActions(input) } catch (cause) { error = cause.message }
  assert.match(error ?? '', mode.includes('source') ? /source budget/ : /prompt exceeds 640 KiB/)
  assert.equal(proposer, mode === 'verifier-prompt' ? 1 : 0)
  assert.equal(verifier, 0)
  const firstCounters = { proposer, verifier }
  const pending = await reviewDraftActions(input)
  const snapshot = pending.registry.articles.find(x => x.date === '2026-10-19')
  assert.equal(snapshot.review.state, 'pending')
  assert.equal(snapshot.verification.proposer, 'not-verified')
  assert.equal(snapshot.verification.verifier, 'not-verified')
  assert.equal(pending.registry.records.length, registry.records.length)
  assert.deepEqual(snapshot.actionKeys, [])
  assert.match(pending.markdown, /Action review pending/)
  results.push({ mode, sourceBytes: sources.map(x => Buffer.byteLength(x.content)), firstInvocationCounters: firstCounters, error, fallback: snapshot.review.state, newIdentities: 0 })
}
assert.equal(JSON.stringify(registry), before)
console.log(JSON.stringify({ sourcePin: pin, method: 'Actual runtime with injected counter agents; no model/provider/network/file writes; all registry changes confined to memory.', results }, null, 2))
```
